function ajouterJoueur(data) { const nom = String(data.nom || "") .trim() .toUpperCase(); const prenom = String(data.prenom || "") .trim(); const categorie = String(data.categorie || "") .trim(); if (!nom || !prenom || !categorie) { return ContentService .createTextOutput( JSON.stringify({ success:false, message:"Informations incomplètes" }) ) .setMimeType(ContentService.MimeType.JSON); } const feuille = SpreadsheetApp .openById(SHEET_ID) .getSheetByName( PLAYERS_SHEET_NAME ); feuille.appendRow([ nom + " " + prenom, categorie, "Oui" ]); return ContentService .createTextOutput( JSON.stringify({ success:true, message:"Joueur ajouté" }) ) .setMimeType(ContentService.MimeType.JSON); }
